🔍 Opérations de Lecture (READ)

Recherche de Livres (Case-Insensitive)
JavaScript - MongoDB Query 
db.books.find({
  $or: [
    { title: { $regex: "orwell", $options: "i" } },
    { author: { $regex: "orwell", $options: "i" } },
    { category: { $regex: "orwell", $options: "i" } }
  ]
})
Python - PyMongo 
def search_books(query):
    db = get_db()
    books = db.books.find({
        "$or": [
            {"title": {"$regex": query, "$options": "i"}},
            {"author": {"$regex": query, "$options": "i"}},
            {"category": {"$regex": query, "$options": "i"}}
        ]
    })
    return list(books)

def get_user_loans(user_id):
    db = get_db()
    user = db.users.find_one({"_id": ObjectId(user_id)})
    loans = user.get('loans', [])
    # Trier par date (plus récent d'abord)
    return sorted(loans, key=lambda x: x['borrow_date'], reverse=True)

✏️ Opérations de Mise à Jour (UPDATE)

Créer un Emprunt (Opération Complexe)

Cette opération met à jour 2 documents de manière atomique.

Étape 1 : Ajouter l'emprunt à l'utilisateur

JavaScript - Opérateur $push et $inc 
db.users.updateOne(
  { _id: ObjectId("user_id") },
  {
    $push: {
      loans: {
        loan_id: new ObjectId(),
        book_id: ObjectId("book_id"),
        book_snapshot: {
          title: "1984",
          author: "George Orwell",
          category: "Classic",
          cover_image: "..."
        },
        borrow_date: new Date(),
        deadline: new Date(Date.now() + 14*24*60*60*1000),
        return_date: null,
        status: "Borrowed"
      }
    },
    $inc: { books_read: 1 }  // Incrémenter compteur
  }
)

Étape 2 : Mettre à jour le livre

JavaScript 
db.books.updateOne(
  { _id: ObjectId("book_id") },
  {
    $push: {
      current_loans: {
        loan_id: loan_id,
        user_id: user_id,
        user_name: "Andro",
        borrow_date: new Date(),
        deadline: new Date(...),
        status: "Borrowed"
      }
    },
    $inc: { available_copies: -1 }  // Diminuer stock
  }
)
Python - Implémentation Complète 
def create_loan(user_id, book_id, deadline):
    db = get_db()
    
    # Vérifier disponibilité
    book = db.books.find_one({
        "_id": ObjectId(book_id),
        "available_copies": {"$gt": 0}
    })
    
    if not book:
        return False, "Book not available"
    
    # ID unique pour le prêt
    loan_id = ObjectId()
    
    # Ajouter à l'utilisateur
    db.users.update_one(
        {"_id": ObjectId(user_id)},
        {
            "$push": {
                "loans": {
                    "loan_id": loan_id,
                    "book_id": book["_id"],
                    "book_snapshot": {...},
                    "borrow_date": datetime.utcnow(),
                    "deadline": deadline,
                    "status": "Borrowed"
                }
            },
            "$inc": {"books_read": 1}
        }
    )
    
    # Mettre à jour le livre
    db.books.update_one(
        {"_id": book["_id"]},
        {
            "$push": {"current_loans": {...}},
            "$inc": {"available_copies": -1}
        }
    )
    
    return True, "Loan created successfully"
✅ 2 opérations atomiques
✅ Pas de transaction complexe

🗑️ Opérations de Suppression (DELETE)

Retourner un Livre

Étape 1 : Mettre à jour le statut dans l'array de l'utilisateur

JavaScript - Positional Operator $ 
db.users.updateOne(
  { "loans.loan_id": ObjectId("loan_id") },
  {
    $set: {
      "loans.$.return_date": new Date(),  // $ = positional operator
      "loans.$.status": "Returned"
    }
  }
)

Étape 2 : Déplacer de current_loans vers loan_history

JavaScript - Opérateurs $pull et $push 
db.books.updateOne(
  { _id: ObjectId("book_id") },
  {
    $pull: { current_loans: { loan_id: ObjectId("loan_id") } },
    $push: { 
      loan_history: {
        loan_id: ObjectId("loan_id"),
        user_name: "Andro",
        borrow_date: ISODate("..."),
        return_date: new Date(),
        status: "Returned"
      }
    },
    $inc: { available_copies: 1 }  // Remettre en stock
  }
)

📊 Requêtes d'Agrégation (Analytics)

Nombre d'Emprunts par Catégorie
JavaScript - Aggregation Pipeline 
db.users.aggregate([
  // Dérouler le tableau loans
  { $unwind: "$loans" },
  
  // Grouper par catégorie
  {
    $group: {
      _id: "$loans.book_snapshot.category",
      count: { $sum: 1 }
    }
  },
  
  // Trier par popularité
  { $sort: { count: -1 } }
])

Python 
def get_loans_by_category():
    db = get_db()
    pipeline = [
        {"$unwind": "$loans"},
        {
            "$group": {
                "_id": "$loans.book_snapshot.category",
                "count": {"$sum": 1}
            }
        },
        {"$sort": {"count": -1}}
    ]
    return list(db.users.aggregate(pipeline))

🔧 Opérateurs MongoDB Utilisés

Opérateur Usage Exemple
$push Ajouter à un array $push: { loans: {...} }
$pull Retirer d'un array $pull: { current_loans: {...} }
$inc Incrémenter/Décrémenter $inc: { available_copies: -1 }
$set Modifier un champ $set: { status: "Late" }
$regex Recherche textuelle { title: { $regex: "...", $options: "i" } }
$ Positional operator $set: { "loans.$.status": "..." }
$unwind Dérouler un array { $unwind: "$loans" }
$group Agréger des données { $group: { _id: "...", count: {...} } }
