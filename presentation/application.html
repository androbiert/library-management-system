Python - Route 
@app.route('/my-loans')
@login_required
def my_loans():
    user = get_user_by_id(session['user_id'])
    loans = user.get('loans', [])
    
    # Séparer actifs et passés
    active_loans = [l for l in loans if l['status'] in ['Borrowed', 'Late']]
    past_loans = [l for l in loans if l['status'] == 'Returned']
    
    return render_template('my_loans.html', 
                         active=active_loans, 
                         past=past_loans)

Python - Badge Attribution Automatique 
def check_and_award_badges(user_id):
    db = get_db()
    user = db.users.find_one({"_id": ObjectId(user_id)})
    
    badges_to_add = []
    current_badges = [b['badge_id'] for b in user.get('badges', [])]
    
    # Bookworm: 5+ livres
    if user['books_read'] >= 5 and 'bookworm' not in current_badges:
        badges_to_add.append({'badge_id': 'bookworm', 'earned_at': datetime.utcnow()})
    
    # Avid Reader: 10+ livres
    if user['books_read'] >= 10 and 'avid_reader' not in current_badges:
        badges_to_add.append({'badge_id': 'avid_reader', 'earned_at': datetime.utcnow()})
    
    # Explorer: 10+ catégories
    if len(user.get('categories_explored', [])) >= 10:
        if 'explorer' not in current_badges:
            badges_to_add.append({'badge_id': 'explorer', 'earned_at': datetime.utcnow()})
    
    if badges_to_add:
        db.users.update_one(
            {"_id": user["_id"]},
            {"$push": {"badges": {"$each": badges_to_add}}}
        )
